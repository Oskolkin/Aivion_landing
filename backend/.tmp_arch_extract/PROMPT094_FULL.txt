type FactualPeriodPair struct {

func ResolveFactFamilyPeriodPair(
    sealed factualapi.FactualReadContext,
    domain string,
    factFamily string,
) (FactualPeriodPair, error) {

    b, ok := sealed.Binding(
        domain,
        factFamily,
    )

    if !ok {
        return FactualPeriodPair{},
            fmt.Errorf(
                "fact family binding missing: %s/%s",
                domain,
                factFamily,
            )
    }

    if b.CoverageTo == nil {
        return FactualPeriodPair{},
            fmt.Errorf(
                "fact family coverage_to missing: %s/%s",
                domain,
                factFamily,
            )
    }

    anchor := dateOnly(*b.CoverageTo)

    return FactualPeriodPair{
        Current: factualapi.Period{
            From: anchor.AddDate(0, 0, -6),
            To:   anchor,
        },
        Previous: factualapi.Period{
            From: anchor.AddDate(0, 0, -13),
            To:   anchor.AddDate(0, 0, -7),
        },
    }, nil
}
